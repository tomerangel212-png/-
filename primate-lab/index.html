לִקְפֹּץ. לְהַקְשִׁיב. לְהַמְצִיא.

מִשְׂחַק לִבְנֵי אָדָם שֶׁסַּקְרָנִים לְגַבֵּי קוֹפִים. בְּכָל תַּחֲנָה מְגַלִּים מַשֶּׁהוּ — וּבַמַּעְבָּדָה הוֹפְכִים אוֹתִיּוֹת לִצְלִילִים.

הַקּוֹף הִמְצִיא כְּלִי. עַכְשָׁו תּוֹרְךָ.

יוֹצְאִים לְגַלּוֹת?

מִשְׁכוּ אֶת הַדְּמוּת לְמַטָּה, שַׁחְרְרוּ וְהֵאָחֲזוּ בַּתַּחֲנָה הַבָּאָה. אֵין פְּסִילָה סוֹפִית — אֶפְשָׁר לְנַסּוֹת שׁוּב.

גַּם בִּישִׁיבָה: כִּוּוּן עוֹזֵר וְאָז שִׁגּוּר. חִצִּים מְשַׁנִּים זָוִית וְעָצְמָה; רֶוַח מְשַׁגֵּר.

הַקְּפִיצוֹת וְהַיְּכֹלוֹת הֵן חֻקֵּי מִשְׂחָק דִּמְיוֹנִיִּים, לֹא הַדְמָיָה זוֹאוֹלוֹגִית. הַלְּמִידָה בַּמָּסָךְ — לֹא חִקּוּי גּוּפָנִי אוֹ מַגָּע עִם חַיּוֹת בַּר.

כָּל אוֹת חֲשׁוּבָה.

הַמָּקוֹר נִשְׁמָר. אַתָּה יָכוֹל לְנַסּוֹת, לְהַקְשִׁיב וּלִשְׁמֹר גִּרְסָה חֲדָשָׁה — בְּלִי לְתַקֵּן אֶת הַמַּחֲשָׁבָה בִּמְקוֹמְךָ.

מֵאוֹת לִצְלִיל

נִסּוּי מוּזִיקָלִי
הַחֲזָרוֹת נִשְׁמָרוֹת. רְוָחִים וְסִמָּנֵי פִּסּוּק מְקַבְּלִים זְמַן; סִמָּנֵי נִקּוּד נִשְׁמָרִים כִּסְמָנִים קְצָרִים. זֶה מִפּוּי מוּזִיקָלִי יְצִירָתִי, לֹא הֲגִיָּה אוֹ תַּרְגּוּם.
הַצְּלִיל מַתְחִיל רַק בִּלְחִיצָה. אֵין הַקְלָטָה וְאֵין גִּישָׁה לַמִּיקְרוֹפוֹן.
מִסְפָּר סִדּוּרִי וְקֹוד לְכָל תָּו
הַכְּלִי הַנִּסְיוֹנִי
לְחִיצָה עַל כַּרִּית מְנַגֶּנֶת אוֹת בּוֹדֶדֶת. הִיא אֵינָהּ מְשַׁנָּה אֶת הָרֶצֶף. גַּם אוֹתִיּוֹת אֲחֵרוֹת מְקַבְּלוֹת צְלִיל, בְּמִפּוּי קָבוּעַ.
הַמְצָאָה, לֹא עוּבְדָּה מֻמְצֵאת.
הַכְּלִי הַמּוּזִיקָלִי הַזֶּה הוּא חֵלֶק מֵהַמִּשְׂחָק שֶׁלָּנוּ. בַּמְּצִיאוּת, יֵשׁ שִׁימְפַּנְזִים שֶׁמְּעַצְּבִים מַקְלוֹת לְהוֹצָאַת טֶרְמִיטִים וּמִשְׁתַּמְּשִׁים בַּאֲבָנִים לְפִצּוּחַ אֱגוֹזִים.

מַחְבֶּרֶת עוֹזֵר הַמַּעְבָּדָה.
מָה גִּלִּיתָ? מָה עוֹד לֹא עָבַד? גַּם הַנִּסּוּי הַקָּטָן שָׁוֶה מָקוֹם.

גִּרְסָאוֹת הַצְּלִיל
הַנְּתוּנִים נִשְׁאָרִים שֶׁלְּךָ.
הַמִּשְׂחָק וְהָרְצָפִים נִשְׁמָרִים בָּאִחְסוּן הַמְּקוֹמִי שֶׁל הַדַּפְדְּפָן. אֵין חֶשְׁבּוֹן, פִּרְסוּמוֹת אוֹ סִנְכְרוּן לֶעָנָן. הַמֵּידָע אֵינוֹ מֻצְפָּן. נִקּוּי נְתוּנֵי הַדַּפְדְּפָן עָלוּל לִמְחֹק אוֹתוֹ; יִצּוּא גִּבּוּי שׁוֹמֵר עוֹתֶק קָרִיא.
בַּיִּבּוּא הַגִּרְסָאוֹת וְהַהֶעָרוֹת מְמֻזָּגוֹת. טְיוּטָה שׁוֹנָה נִשְׁמֶרֶת כְּגִרְסָה; הַטְּיוּטָה וְהַמַּסָּע הַנּוֹכְחִיִּים לֹא נִדְרָסִים.
מְקוֹרוֹת, דִּיּוּק וְהַפְרָדָה מֵהַדִּמְיוֹן
גּוֹרִילוֹת וְשִׁימְפַּנְזִים הֵם קוֹפֵי־אָדָם לְלֹא זָנָב; בָּבּוּנִים הֵם קוֹפִים בַּעֲלֵי זָנָב שֶׁאֵינוֹ לוֹפֵת. הַדְּמוּיוֹת הַמְּאֻיָּרוֹת הֵן הַפְשָׁטָה מִשְׂחָקִית.
גַּן הַחַיּוֹת שֶׁל סָן דִּיֵּיגוֹ: שִׁימְפַּנְזָהגַּן הַחַיּוֹת שֶׁל סָן דִּיֵּיגוֹ: גּוֹרִילָהגַּן הַחַיּוֹת שֶׁל סָן דִּיֵּיגוֹ: בָּבּוּן מִצְרִימְכוֹן גֵ׳יין גּוּדְאוֹל: מָסוֹרוֹת וְשִׁמּוּשׁ בְּכֵלִיםתִּעוּד הָאִחְסוּן הַמְּקוֹמִי בַּדַּפְדְּפָן
זֶהוּ מִשְׂחַק תרא מְקוֹרִי. הוּא אֵינוֹ הַמִּשְׂחָק הַמִּסְחָרִי ״סְלִינְג קוֹנְג״, אֵינוֹ מְסֻנָּף אֵלָיו וְאֵינוֹ מִשְׁתַּמֵּשׁ בַּקּוֹד, בַּדְּמוּיוֹת אוֹ בַּצְּלִילִים שֶׁלּוֹ.
